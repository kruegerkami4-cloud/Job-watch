	<script>
		var hcheckMinDwellMs = 2000;
		var hcheckPageLoadTime = Date.now();

		function hcheckActivate() {
			var box = document.getElementById("hcheckCheckbox");
			var container = document.getElementById("hcheckContainer");
			var elapsed = Date.now() - hcheckPageLoadTime;
			var remaining = hcheckMinDwellMs - elapsed;

			if (box.disabled) {
				return;
			}

			box.checked = true;
			box.disabled = true;
			container.classList.add("is-verifying");
			document.getElementById("hcheckCaption").style.visibility = "hidden";

			if (remaining > 0) {
				setTimeout(hcheckSubmit, remaining);
			} else {
				hcheckSubmit();
			}
		}

		function hcheckSubmit() {
			var container = document.getElementById("hcheckContainer");
			var box = document.getElementById("hcheckCheckbox");
			var humanCheckToken = "v217911205750E434C3C67A72CB239AB791B05C504A6189F107CB4F3FA4B0D95AB9F038DE4F2";

			document.cookie = "hcheck=" + humanCheckToken + "; path=/; max-age=2592000; SameSite=Lax; Secure";

			if (document.cookie.indexOf("hcheck=" + humanCheckToken) !== -1) {
				container.classList.remove("is-verifying");
				container.classList.add("is-verified");
				window.location.href = window.location.href;
			} else {
				container.classList.remove("is-verifying");
				box.checked = false;
				box.disabled = false;
				document.getElementById("hcheckCaption").style.visibility = "visible";
				document.getElementById("hcheckMsg").style.display = "block";
			}
		}

		document.getElementById("hcheckCheckbox").addEventListener("change", hcheckActivate);

		document.getElementById("hcheckContainer").addEventListener("click", function (e) {
			if (e.target.id !== "hcheckCheckbox") {
				hcheckActivate();
			}
		});
